cmake_minimum_required(VERSION 3.16)
project(offline_asr LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SHERPA_ROOT "" CACHE PATH "Extracted sherpa-onnx distribution directory")
find_path(SHERPA_INCLUDE sherpa-onnx/c-api/c-api.h PATHS "${SHERPA_ROOT}/include" NO_DEFAULT_PATH)
if(WIN32 AND MINGW)
  find_file(SHERPA_LIBRARY sherpa-onnx-c-api.dll PATHS "${SHERPA_ROOT}/lib" "${SHERPA_ROOT}/bin" NO_DEFAULT_PATH)
else()
  find_library(SHERPA_LIBRARY NAMES sherpa-onnx-c-api PATHS "${SHERPA_ROOT}/lib" NO_DEFAULT_PATH)
endif()
if(NOT SHERPA_INCLUDE OR NOT SHERPA_LIBRARY)
  message(FATAL_ERROR "Set SHERPA_ROOT to an extracted distribution containing include/ and lib/.")
endif()
add_library(wav STATIC src/wav.c)
target_include_directories(wav PUBLIC src)
add_executable(asr_demo src/main.c src/asr.c)
target_include_directories(asr_demo PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(asr_demo PRIVATE wav "${SHERPA_LIBRARY}")
get_filename_component(SHERPA_LIB_DIR "${SHERPA_LIBRARY}" DIRECTORY)
set_target_properties(asr_demo PROPERTIES BUILD_RPATH "${SHERPA_LIB_DIR}")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(wav PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(asr_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  file(GLOB SHERPA_DLLS "${SHERPA_ROOT}/bin/*.dll" "${SHERPA_ROOT}/lib/*.dll")
  foreach(DLL IN LISTS SHERPA_DLLS)
    add_custom_command(TARGET asr_demo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" "$<TARGET_FILE_DIR:asr_demo>")
  endforeach()
endif()
enable_testing()
add_executable(test_wav tests/test_wav.c)
target_link_libraries(test_wav PRIVATE wav)
add_test(NAME wav_format_validation COMMAND test_wav)
add_test(NAME cli_help COMMAND asr_demo --help)
if(UNIX AND EXISTS "${CMAKE_SOURCE_DIR}/models/sherpa-onnx-sense-voice-zh-en-ja-ko-yue-int8-2024-07-17/model.int8.onnx")
  add_test(NAME cli_error_scenarios COMMAND sh "${CMAKE_SOURCE_DIR}/tests/test_cli.sh" "$<TARGET_FILE:asr_demo>")
  set_tests_properties(cli_error_scenarios PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_test(NAME real_model_smoke COMMAND asr_demo
    --model-dir "${CMAKE_SOURCE_DIR}/models/sherpa-onnx-sense-voice-zh-en-ja-ko-yue-int8-2024-07-17"
    --wav "${CMAKE_SOURCE_DIR}/models/sherpa-onnx-sense-voice-zh-en-ja-ko-yue-int8-2024-07-17/test_wavs/zh.wav")
  set_tests_properties(real_model_smoke PROPERTIES PASS_REGULAR_EXPRESSION "开饭时间早上9点至下午5点")
endif()
